async function startCamera() { try { statusDiv.textContent = 'מבקש גישה למצלמה...'; // שליפת רשימת המצלמות המחוברות למחשב const devices = await navigator.mediaDevices.enumerateDevices(); const videoDevices = devices.filter(device => device.kind === 'videoinput'); let constraints = { video: true, audio: true }; // אם נמצאה מצלמה ספציפית, נגדיר אותה ישירות if (videoDevices.length > 0) { constraints.video = { deviceId: { exact: videoDevices[0].deviceId } }; } const stream = await navigator.mediaDevices.getUserMedia(constraints); localVideo.srcObject = stream; await localVideo.play(); statusDiv.textContent = 'המצלמה פועלת בהצלחה!'; } catch (error) { console.error('Camera error:', error); statusDiv.textContent = 'שגיאה: מצלמה לא זמינה'; alert('הדפדפן לא מצליח לתפוס את המצלמה. נסה ללחוץ על "איפוס הרשאות" (הכפתור שבתמונה ששלחת), רענן את העמוד ונסה שוב.'); } }
